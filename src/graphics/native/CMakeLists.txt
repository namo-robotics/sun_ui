cmake_minimum_required(VERSION 3.22)
project(sun_ui_graphics LANGUAGES C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GL REQUIRED IMPORTED_TARGET epoxy glib-2.0)
add_library(sun_ui_graphics STATIC graphics.c)
target_link_libraries(sun_ui_graphics PRIVATE PkgConfig::GL)
target_compile_options(sun_ui_graphics PRIVATE -Wall -Wextra -Werror)
option(BUILD_SDL_HOST "Build the separate SDL3 window host" OFF)
if(BUILD_SDL_HOST)
  pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl3>=3.2)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_file(GU_CAIRO_PLUGIN libdecor-cairo.so PATHS "/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}/libdecor/plugins-1" "/usr/lib64/libdecor/plugins-1" "/usr/lib/libdecor/plugins-1")
    if(GU_CAIRO_PLUGIN)
      file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/../libdecor")
      file(CREATE_LINK "${GU_CAIRO_PLUGIN}" "${CMAKE_BINARY_DIR}/../libdecor/libdecor-cairo.so" SYMBOLIC)
    else()
      message(STATUS "Cairo libdecor plugin not found; Wayland requires a Cairo-only LIBDECOR_PLUGIN_DIR at runtime")
    endif()
  endif()
  add_library(sun_ui_sdl STATIC ../../adapters/sdl/native.c)
  target_link_libraries(sun_ui_sdl PRIVATE sun_ui_graphics PkgConfig::SDL PkgConfig::GL)
  target_compile_options(sun_ui_sdl PRIVATE -Wall -Wextra -Werror)
  option(BUILD_GRAPHICS_TESTS "Build native event injection for API acceptance tests" OFF)
  if(BUILD_GRAPHICS_TESTS)
    add_library(graphics_test_input STATIC ../../../tests/graphics/input.c)
    target_link_libraries(graphics_test_input PRIVATE PkgConfig::SDL)
    target_compile_options(graphics_test_input PRIVATE -Wall -Wextra -Werror)
  endif()
endif()
